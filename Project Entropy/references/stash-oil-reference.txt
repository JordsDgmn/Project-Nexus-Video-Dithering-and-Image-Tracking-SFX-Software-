"use client";

import { useEffect, useRef } from "react";

/** Oil: a thin film of iridescent color sliding over a dark surface, swirling under the pointer and rippling on click. WebGL fragment shader, no dependencies. MIT, from stash.style/effects. */
export type OilProps = {
  /** Color of the dark surface under the film. Default "#020208". */
  background?: string;
  /** Color washed through the iridescence. Default "#7C3AED". */
  tint?: string;
  /** How fast the effect moves; 0 freezes it. Default 1. */
  speed?: number;
  /** Size of the swirls; higher packs more in. Default 1. */
  scale?: number;
  /** How many color bands the film shows. Default 1.6. */
  bands?: number;
  /** How vivid the film colors are. Default 0.9. */
  saturation?: number;
  /** Strength of the white highlights on the film. Default 0.7. */
  sheen?: number;
  /** How much of the dark surface shows through. Default 0.7. */
  darkness?: number;
  /** Lets the effect respond to the pointer. Default true. */
  mouseInteraction?: boolean;
  /** Sends a ripple out from every click. Default true. */
  clickRipple?: boolean;
  className?: string;
};

const DEFAULTS = {
  background: "#020208",
  tint: "#7C3AED",
  speed: 1,
  scale: 1,
  bands: 1.6,
  saturation: 0.9,
  sheen: 0.7,
  darkness: 0.7,
  mouseInteraction: true,
  clickRipple: true,
};

const FRAG = `precision highp float;
uniform vec2 u_res;
uniform float u_time;
uniform float u_now;
uniform float u_dpr;
uniform vec2 u_mouse;
uniform float u_hover;
uniform vec3 u_clicks[4];
uniform vec3 u_background;
uniform vec3 u_tint;
uniform float u_speed;
uniform float u_scale;
uniform float u_bands;
uniform float u_saturation;
uniform float u_sheen;
uniform float u_darkness;
uniform float u_mouseInteraction;
uniform float u_clickRipple;
float h21(vec2 p){p=fract(p*vec2(123.34,456.21));p+=dot(p,p+45.32);return fract(p.x*p.y);}
float vn(vec2 p){vec2 i=floor(p),f=fract(p);vec2 u=f*f*(3.-2.*f);return mix(mix(h21(i),h21(i+vec2(1.,0.)),u.x),mix(h21(i+vec2(0.,1.)),h21(i+vec2(1.,1.)),u.x),u.y);}
float fbm(vec2 p){float s=0.,a=.5;mat2 r=mat2(.8,.6,-.6,.8);for(int i=0;i<5;i++){s+=a*vn(p);p=r*p*2.03+vec2(1.7,9.2);a*=.5;}return s;}
float sn(vec2 p){return vn(p)*.65+vn(p*2.07+vec2(3.1,7.7))*.35;}
vec2 P(){return (gl_FragCoord.xy-.5*u_res)/u_res.y;}
vec2 M(){return (u_mouse-.5)*vec2(u_res.x/u_res.y,1.);}
mat2 rot(float a){float c=cos(a),s=sin(a);return mat2(c,-s,s,c);}
float ripple(vec2 uv,float speed,float width){float s=0.;for(int i=0;i<4;i++){vec3 c=u_clicks[i];float age=u_now-c.z;if(age>=0.&&age<2.5){vec2 cp=(c.xy-.5)*vec2(u_res.x/u_res.y,1.);float d=abs(length(uv-cp)-age*speed);s+=exp(-d*d/(width*width))*(1.-age/2.5);}}return s;}
void main(){
  float t=u_time;
  vec2 uv0=P();
  vec2 uv=uv0*u_scale;
  vec2 q=vec2(fbm(uv*1.2+t*.08),fbm(uv*1.2+vec2(5.2,1.3)-t*.07));
  vec2 r=vec2(fbm(uv*1.4+3.*q+vec2(1.7,9.2)+t*.05),fbm(uv*1.4+3.*q+vec2(8.3,2.8)-t*.06));
  float h=fbm(uv*.9+3.*r)*.6+sn(uv*.7+2.*r)*.4;
  vec2 dm=uv0-M();
  float inf=exp(-dot(dm,dm)*10.)*u_hover*u_mouseInteraction;
  h+=inf*.22*sin(length(dm)*36.-u_now*4.);
  h+=ripple(uv0,.5,.05)*u_clickRipple*.2;
  float ph=h*u_bands*3.+t*.05;
  vec3 col=.5+.5*cos(6.28318*(ph+vec3(0.,.33,.67)));
  col=mix(vec3(dot(col,vec3(.333))),col,u_saturation);
  col=mix(col,col*u_tint*2.2,.3);
  float th=u_darkness*.7;
  float cover=smoothstep(th,th+.28,sn(uv*.8+r*1.5-t*.03+9.));
  float band=.72+.28*cos(ph*12.566);
  vec3 o=mix(u_background,col*band,cover);
  vec3 n=normalize(vec3((r-.5)*2.4,1.));
  float spec=pow(max(dot(n,normalize(vec3(.35,.5,1.))),0.),28.)*u_sheen*.7;
  o+=vec3(spec)*cover;
  o+=(h21(gl_FragCoord.xy+fract(u_now*7.)*113.)-.5)*.035;
  gl_FragColor=vec4(o,1.);
}
`;

const VERT = "attribute vec2 p;void main(){gl_Position=vec4(p,0.,1.);}";

const rgb = (hex: string): [number, number, number] => {
  const h = hex.replace("#", "");
  const full = h.length === 3 ? h.split("").map((c) => c + c).join("") : h.padEnd(6, "0").slice(0, 6);
  const n = parseInt(full, 16);
  return [((n >> 16) & 255) / 255, ((n >> 8) & 255) / 255, (n & 255) / 255];
};

const defined = <T extends object>(o: T): Partial<T> =>
  Object.fromEntries(Object.entries(o).filter(([, v]) => v !== undefined)) as Partial<T>;

export default function Oil({ className, ...props }: OilProps) {
  const ref = useRef<HTMLDivElement>(null);
  const live = useRef(props);
  const kick = useRef<() => void>(() => {});
  // Props flow into uniforms every frame, so changing one never rebuilds the GL context.
  useEffect(() => {
    live.current = props;
    kick.current();
  });
  useEffect(() => {
    const host = ref.current;
    if (!host) return;
    // A fresh canvas per mount: a context lost in cleanup is never handed back on remount.
    const canvas = document.createElement("canvas");
    canvas.style.cssText = "position:absolute;inset:0;width:100%;height:100%;display:block";
    host.appendChild(canvas);
    const gl = canvas.getContext("webgl", { antialias: false, premultipliedAlpha: false });
    if (!gl) { canvas.remove(); return; }
    const dpr = Math.min(window.devicePixelRatio || 1, 1.5);
    const still = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    const compile = (type: number, src: string) => {
      const s = gl.createShader(type)!;
      gl.shaderSource(s, src);
      gl.compileShader(s);
      return s;
    };
    const prog = gl.createProgram()!;
    gl.attachShader(prog, compile(gl.VERTEX_SHADER, VERT));
    gl.attachShader(prog, compile(gl.FRAGMENT_SHADER, FRAG));
    gl.linkProgram(prog);
    gl.useProgram(prog);
    gl.bindBuffer(gl.ARRAY_BUFFER, gl.createBuffer());
    gl.bufferData(gl.ARRAY_BUFFER, new Float32Array([-1, -1, 3, -1, -1, 3]), gl.STATIC_DRAW);
    const loc = gl.getAttribLocation(prog, "p");
    gl.enableVertexAttribArray(loc);
    gl.vertexAttribPointer(loc, 2, gl.FLOAT, false, 0, 0);
    const u = (n: string) => gl.getUniformLocation(prog, n);
    const U = {
      res: u("u_res"), time: u("u_time"), now: u("u_now"), dpr: u("u_dpr"),
      mouse: u("u_mouse"), hover: u("u_hover"), clicks: u("u_clicks"),
      background: u("u_background"),
      tint: u("u_tint"),
      speed: u("u_speed"),
      scale: u("u_scale"),
      bands: u("u_bands"),
      saturation: u("u_saturation"),
      sheen: u("u_sheen"),
      darkness: u("u_darkness"),
      mouseInteraction: u("u_mouseInteraction"),
      clickRipple: u("u_clickRipple"),
    };

    const ptr = { x: 0.5, y: 0.5, tx: 0.5, ty: 0.5, hover: 0, inside: false };
    const clicks = new Float32Array(12).fill(-100);
    let nextClick = 0;
    let raf = 0;
    let visible = true;
    let time = 10;
    const start = performance.now();
    let last = start;
    const resize = () => {
      canvas.width = Math.max(1, Math.round(canvas.clientWidth * dpr));
      canvas.height = Math.max(1, Math.round(canvas.clientHeight * dpr));
      gl.viewport(0, 0, canvas.width, canvas.height);
    };
    const frame = (now: number) => {
      raf = 0;
      const p = { ...DEFAULTS, ...defined(live.current) };
      const dt = Math.min(0.05, Math.max(0, (now - last) / 1000));
      last = now;
      time += dt * p.speed;
      ptr.x += (ptr.tx - ptr.x) * 0.1;
      ptr.y += (ptr.ty - ptr.y) * 0.1;
      ptr.hover += ((ptr.inside ? 1 : 0) - ptr.hover) * 0.06;
      gl.uniform2f(U.res, canvas.width, canvas.height);
      gl.uniform1f(U.time, time);
      gl.uniform1f(U.now, (now - start) / 1000);
      gl.uniform1f(U.dpr, dpr);
      gl.uniform2f(U.mouse, ptr.x, ptr.y);
      gl.uniform1f(U.hover, ptr.hover);
      gl.uniform3fv(U.clicks, clicks);
      gl.uniform3fv(U.background, rgb(p.background));
      gl.uniform3fv(U.tint, rgb(p.tint));
      gl.uniform1f(U.speed, p.speed);
      gl.uniform1f(U.scale, p.scale);
      gl.uniform1f(U.bands, p.bands);
      gl.uniform1f(U.saturation, p.saturation);
      gl.uniform1f(U.sheen, p.sheen);
      gl.uniform1f(U.darkness, p.darkness);
      gl.uniform1f(U.mouseInteraction, p.mouseInteraction ? 1 : 0);
      gl.uniform1f(U.clickRipple, p.clickRipple ? 1 : 0);
      gl.drawArrays(gl.TRIANGLES, 0, 3);
      if (visible && !still) raf = requestAnimationFrame(frame);
    };
    kick.current = () => {
      if (!raf) raf = requestAnimationFrame(frame);
    };
    const locate = (e: PointerEvent) => {
      const r = canvas.getBoundingClientRect();
      const x = (e.clientX - r.left) / r.width;
      const y = 1 - (e.clientY - r.top) / r.height;
      return { x, y, inside: x >= 0 && x <= 1 && y >= 0 && y <= 1 };
    };
    const move = (e: PointerEvent) => {
      const l = locate(e);
      if (l.inside && ptr.hover < 0.02) { ptr.x = l.x; ptr.y = l.y; }
      ptr.tx = l.x; ptr.ty = l.y; ptr.inside = l.inside;
    };
    const down = (e: PointerEvent) => {
      const l = locate(e);
      if (!l.inside) return;
      clicks.set([l.x, l.y, (performance.now() - start) / 1000], nextClick * 3);
      nextClick = (nextClick + 1) % 4;
    };
    const leave = () => { ptr.inside = false; };
    const ro = new ResizeObserver(() => { resize(); kick.current(); });
    ro.observe(canvas);
    resize();
    const io = new IntersectionObserver(([e]) => {
      visible = e.isIntersecting;
      if (visible) kick.current();
    });
    io.observe(canvas);
    window.addEventListener("pointermove", move, { passive: true });
    window.addEventListener("pointerdown", down, { passive: true });
    document.addEventListener("pointerleave", leave);
    kick.current();
    return () => {
      cancelAnimationFrame(raf);
      kick.current = () => {};
      ro.disconnect();
      io.disconnect();
      window.removeEventListener("pointermove", move);
      window.removeEventListener("pointerdown", down);
      document.removeEventListener("pointerleave", leave);
      gl.getExtension("WEBGL_lose_context")?.loseContext();
      canvas.remove();
    };
  }, []);
  return <div ref={ref} className={className} aria-hidden="true" style={{ position: "absolute", inset: 0, overflow: "hidden" }} />;
}
